Defines the shared props interface for team sub-navigation components, providing typed access to routing context and location state.

## Key Components

- **`ITeamSubnavProps`** — Interface for team subnav component props, including:
  - `location` — Current route location with `pathname`, `search`, `hash`, and a `query` object supporting an optional `fleet_id` parameter
  - `router` — Injected React Router instance for programmatic navigation

## Usage Example

```typescript
import { ITeamSubnavProps } from "./team_subnav";

const TeamSubnavComponent: React.FC<ITeamSubnavProps> = ({ location, router }) => {
  const { fleet_id } = location.query;

  const handleNavigate = () => {
    router.push(`/teams?fleet_id=${fleet_id}`);
  };

  return <nav onClick={handleNavigate}>Team Navigation</nav>;
};
```
